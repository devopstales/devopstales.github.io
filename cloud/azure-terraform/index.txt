In this post I will show you how how you can create a VM in Azure with Terraform.

&lt;!--more--&gt;

With Azure we can use Azure blob storage as a shared storage for the Terraform states. So firs we need to create a storage account for this purpose:

```bash
az group create --name tf-state-rg \
  --location westeurope

export STORAGE_ACCOUNT_NAME=tfstatesa$(dd bs=4 count=1 if=/dev/urandom 2&gt;/dev/null | xxd -p)

echo $STORAGE_ACCOUNT_NAME
tfstatesa273b9cdf

az storage account create --name $STORAGE_ACCOUNT_NAME \
  --location westeurope \
  --resource-group tf-state-rg

az storage container create --account-name $STORAGE_ACCOUNT_NAME \
  --name tfstate \
  --public-access off

az storage account key list --resource-group tf-state-rg --account-name $STORAGE_ACCOUNT_NAME
```

Create provider config:

```json
nano providers.tf
terraform {
  required_version = &#34;&gt;=0.12&#34;

  required_providers {
    azurerm = {
      source  = &#34;hashicorp/azurerm&#34;
      version = &#34;~&gt;2.0&#34;
    }
  }

  backend &#34;azurerm&#34; {
    resource_group_name  = &#34;tf-state-rg&#34;
    storage_account_name = &#34;tfstatesa273b9cdf&#34;
    container_name       = &#34;tfstate&#34;
    key                  = &#34;terraform.tfstate&#34;
  }

}

provider &#34;azurerm&#34; {
  features {}
}
```

Create output config:

```json
nano outputs.tf
output &#34;resource_group_name&#34; {
  value = azurerm_resource_group.rg.name
}

output &#34;public_ip_address&#34; {
  value = azurerm_linux_virtual_machine.my_terraform_vm.public_ip_address
}
```

Create variable config:

```json
nano variables.tf
variable &#34;resource_group_location&#34; {
  type        = string
  default     = &#34;westeurope&#34;
  description = &#34;Location of the resource group.&#34;
}

variable &#34;resource_group_name&#34; {
  type        = string
  default     = &#34;egis-terraform-demo&#34;
  description = &#34;Name of the resource group.&#34;
}


variable &#34;ssh_key&#34; {
  default = &#34;ssh-rsa AAAAB3NzaC1yc2EAAAADAQABA...&#34;
}
```

Create the main Terraform config file vit the resource definitions:

```json
nano main.tf
resource &#34;azurerm_resource_group&#34; &#34;rg&#34; {
  location = var.resource_group_location
  name     = var.resource_group_name
}

# Create virtual network
resource &#34;azurerm_virtual_network&#34; &#34;my_terraform_network&#34; {
  name                = &#34;myVnet&#34;
  address_space       = [&#34;10.0.0.0/16&#34;]
  location            = azurerm_resource_group.rg.location
  resource_group_name = azurerm_resource_group.rg.name
}

# Create subnet
resource &#34;azurerm_subnet&#34; &#34;my_terraform_subnet&#34; {
  name                 = &#34;mySubnet&#34;
  resource_group_name  = azurerm_resource_group.rg.name
  virtual_network_name = azurerm_virtual_network.my_terraform_network.name
  address_prefixes     = [&#34;10.0.1.0/24&#34;]
}

# Create public IPs
resource &#34;azurerm_public_ip&#34; &#34;my_terraform_public_ip&#34; {
  name                = &#34;myPublicIP&#34;
  location            = azurerm_resource_group.rg.location
  resource_group_name = azurerm_resource_group.rg.name
  allocation_method   = &#34;Dynamic&#34;
}

# Create Network Security Group and rule
resource &#34;azurerm_network_security_group&#34; &#34;my_terraform_nsg&#34; {
  name                = &#34;myNetworkSecurityGroup&#34;
  location            = azurerm_resource_group.rg.location
  resource_group_name = azurerm_resource_group.rg.name

  security_rule {
    name                       = &#34;SSH&#34;
    priority                   = 1001
    direction                  = &#34;Inbound&#34;
    access                     = &#34;Allow&#34;
    protocol                   = &#34;Tcp&#34;
    source_port_range          = &#34;*&#34;
    destination_port_range     = &#34;22&#34;
    source_address_prefix      = &#34;*&#34;
    destination_address_prefix = &#34;*&#34;
  }
}

# Create network interface
resource &#34;azurerm_network_interface&#34; &#34;my_terraform_nic&#34; {
  name                = &#34;myNIC&#34;
  location            = azurerm_resource_group.rg.location
  resource_group_name = azurerm_resource_group.rg.name

  ip_configuration {
    name                          = &#34;my_nic_configuration&#34;
    subnet_id                     = azurerm_subnet.my_terraform_subnet.id
    private_ip_address_allocation = &#34;Dynamic&#34;
    public_ip_address_id          = azurerm_public_ip.my_terraform_public_ip.id
  }
}

# Connect the security group to the network interface
resource &#34;azurerm_network_interface_security_group_association&#34; &#34;example&#34; {
  network_interface_id      = azurerm_network_interface.my_terraform_nic.id
  network_security_group_id = azurerm_network_security_group.my_terraform_nsg.id
}

# Generate random text for a unique storage account name
resource &#34;random_id&#34; &#34;random_id&#34; {
  keepers = {
    # Generate a new ID only when a new resource group is defined
    resource_group = azurerm_resource_group.rg.name
  }

  byte_length = 8
}

# Create storage account for boot diagnostics
resource &#34;azurerm_storage_account&#34; &#34;my_storage_account&#34; {
  name                     = &#34;diag${random_id.random_id.hex}&#34;
  location                 = azurerm_resource_group.rg.location
  resource_group_name      = azurerm_resource_group.rg.name
  account_tier             = &#34;Standard&#34;
  account_replication_type = &#34;LRS&#34;
}

# Create virtual machine
resource &#34;azurerm_linux_virtual_machine&#34; &#34;my_terraform_vm&#34; {
  name                  = &#34;myVM&#34;
  location              = azurerm_resource_group.rg.location
  resource_group_name   = azurerm_resource_group.rg.name
  network_interface_ids = [azurerm_network_interface.my_terraform_nic.id]
  size                  = &#34;Standard_DS1_v2&#34;

  os_disk {
    name                 = &#34;myOsDisk&#34;
    caching              = &#34;ReadWrite&#34;
    storage_account_type = &#34;Premium_LRS&#34;
  }

  source_image_reference {
    publisher = &#34;Canonical&#34;
    offer     = &#34;0001-com-ubuntu-server-jammy&#34;
    sku       = &#34;22_04-lts-gen2&#34;
    version   = &#34;latest&#34;
  }

  computer_name                   = &#34;myvm&#34;
  admin_username                  = &#34;azureuser&#34;
  disable_password_authentication = true

  admin_ssh_key {
    username   = &#34;azureuser&#34;
    public_key = var.ssh_key
  }

  boot_diagnostics {
    storage_account_uri = azurerm_storage_account.my_storage_account.primary_blob_endpoint
  }
}
```

Now we can create the VM:

```bash
az login

# init modules
terraform init -upgrade

# plan creation
terraform plan -out main.tfplan

# apply plan
terraform apply main.tfplan

# cleanup
terraform plan -destroy -out main.destroy.tfplan
terraform apply main.destroy.tfplan
```
